class Encryption:
    def __init__(self):
        self.key = self.generate_key()
    
    def generate_key(self):
        machine_guid = str(uuid.getnode())
        return hashlib.sha256(machine_guid.encode()).digest()
    
    def encrypt(self, data):
        try:
            if isinstance(data, str):
                data = data.encode()
            iv = get_random_bytes(16)
            cipher = AES.new(self.key, AES.MODE_CBC, iv)
            encrypted = cipher.encrypt(pad(data, AES.block_size))
            return base64.b64encode(iv + encrypted).decode()
        except:
            return data
    
    def decrypt(self, encrypted_data):
        try:
            if isinstance(encrypted_data, str):
                encrypted_data = base64.b64decode(encrypted_data.encode())
            iv = encrypted_data[:16]
            cipher = AES.new(self.key, AES.MODE_CBC, iv)
            decrypted = unpad(cipher.decrypt(encrypted_data[16:]), AES.block_size)
            return decrypted.decode()
        except:
            return encrypted_data

def protect_console():
    if os.name == 'nt':
        try:
            kernel32 = ctypes.windll.kernel32
            user32 = ctypes.windll.user32
            console_handle = kernel32.GetConsoleWindow()
            current_style = user32.GetWindowLongA(console_handle, -16)
            user32.SetWindowLongA(console_handle, -16, current_style & ~0x00F00000)
            input_handle = kernel32.GetStdHandle(-10)
            current_mode = ctypes.c_uint32()
            kernel32.GetConsoleMode(input_handle, ctypes.byref(current_mode))
            new_mode = current_mode.value & ~0x0040
            new_mode = new_mode | 0x0080
            kernel32.SetConsoleMode(input_handle, new_mode)
        except Exception as error:
            pass

def get_center_pixel():
    try:
        user32 = ctypes.windll.user32
        screen_width, screen_height = user32.GetSystemMetrics(0), user32.GetSystemMetrics(1)
        center_x = screen_width // 2
        center_y = screen_height // 2
        return center_x, center_y
    except:
        return 960, 540

def trigger_shot():
    global last_shot_timestamp
    current_time = time.time()
    if current_time - last_shot_timestamp >= SHOT_DELAY:
        try:
            win32api.mouse_event(win32con.MOUSEEVENTF_LEFTDOWN, 0, 0, 0, 0)
            time.sleep(0.001)
            win32api.mouse_event(win32con.MOUSEEVENTF_LEFTUP, 0, 0, 0, 0)
            last_shot_timestamp = current_time
        except:

def get_key_name(vk_code):
    key_names = {
        0x01: "left", 0x02: "right", 0x04: "middle",
        0x05: "xbutton1", 0x06: "xbutton2",
        0x10: "shift", 0x11: "ctrl", 0x12: "alt",
        0x14: "capslock", 0x20: "space", 0x25: "left", 
        0x26: "up", 0x27: "right", 0x28: "down"
    }
    return key_names.get(vk_code, f"key_{vk_code}")

def wait_for_key_input():
    global current_hold_key, waiting_for_key_input
    
    print(f"{Fore.YELLOW}Press any key or mouse button...")
    
    time.sleep(0.5)
    
    pressed_keys = set()
    start_time = time.time()
    
    while waiting_for_key_input and time.time() - start_time < 10:
        try:
            for vk_code in range(1, 256):
                if win32api.GetAsyncKeyState(vk_code) & 0x8000:
                    if vk_code not in pressed_keys:
                        pressed_keys.add(vk_code)
                        if vk_code == 0x1B:
                            return None
                        time.sleep(0.3)
                        return get_key_name(vk_code)
            
            for mouse_code, button_name in MOUSE_CODES.items():
                if win32api.GetAsyncKeyState(mouse_code) & 0x8000:
                    if mouse_code not in pressed_keys:
                        pressed_keys.add(mouse_code)
                        time.sleep(0.3)
                        return button_name
            
            time.sleep(0.01)
        except:
            time.sleep(0.01)
    
    return None
